```python
"""
src/00_build_universe.py — Build KOSPI firm universe from Bloomberg.

OUTPUT: data/raw/universe_raw.csv
COLUMNS: ticker, name, sector, industry, country, ipo_date

Run from project root:
    python src/00_build_universe.py

Requires blpapi (Bloomberg terminal only):
    pip install blpapi
"""

import os
import sys
import logging
from datetime import datetime

import pandas as pd

# --- sys.path fix: ensure project root is on path so utils/ is importable ---
# Required because "python src/script.py" adds src/ to sys.path, not CWD.
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

# --- Graceful blpapi guard ---
try:
    from utils.bbg import bds, bdp
except RuntimeError:
    # utils.bbg imports fine without blpapi; RuntimeError only raised on call
    from utils.bbg import bds, bdp

# Check blpapi availability before doing any work
try:
    import blpapi  # noqa: F401
except ImportError:
    print(
        "Error: blpapi is not installed.\n"
        "This script must be run at a Bloomberg terminal:\n"
        "  pip install blpapi\n"
        "  python src/00_build_universe.py",
        file=sys.stderr,
    )
    sys.exit(1)

# --- Logging ---
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s  %(levelname)s  %(message)s",
    datefmt="%H:%M:%S",
)
log = logging.getLogger(__name__)

# --- Constants ---
OUTPUT_PATH = os.path.join("data", "raw", "universe_raw.csv")
KOSPI_INDEX = "KOSPI Index"
MEMBERS_FIELD = "INDX_MEMBERS"
IDENTIFYING_FIELDS = [
    "TICKER",
    "NAME",
    "GICS_SECTOR_NAME",
    "GICS_INDUSTRY_NAME",
    "CNTRY_ISSUE_ISO",
    "EQY_FUND_DT",  # IPO / listing date — used to drop post-2023 firms
]
IPO_CUTOFF = datetime(2023, 1, 1)


def main():
    log.info("Step 1: Pulling KOSPI members via BDS(%r, %r)", KOSPI_INDEX, MEMBERS_FIELD)
    members = bds(KOSPI_INDEX, MEMBERS_FIELD)
    log.info("  Found %d raw KOSPI members.", len(members))

    if not members:
        log.error(
            "BDS returned empty list. Possible cause: wrong INDX_MEMBERS sub-element key.\n"
            "See utils/bbg.py bds() — it logs sub-element names on stderr when the key fails.\n"
            "Check the logged element names and update bbg.bds() accordingly."
        )
        sys.exit(1)

    log.info("Step 2: Pulling identifying fields via BDP for %d tickers.", len(members))
    df = bdp(members, IDENTIFYING_FIELDS)
    df.index.name = "bloomberg_ticker"
    df.reset_index(inplace=True)
    log.info("  BDP returned %d rows.", len(df))

    # Rename for output clarity
    df = df.rename(columns={
        "bloomberg_ticker": "bloomberg_ticker",
        "TICKER": "ticker",
        "NAME": "name",
        "GICS_SECTOR_NAME": "sector",
        "GICS_INDUSTRY_NAME": "industry",
        "CNTRY_ISSUE_ISO": "country",
        "EQY_FUND_DT": "ipo_date",
    })

    # Step 3: Drop financials (GICS sector = "Financials")
    pre_filter = len(df)
    df = df[df["sector"] != "Financials"]
    log.info(
        "Step 3: Dropped %d financial firms (sector='Financials'). Remaining: %d.",
        pre_filter - len(df), len(df),
    )

    # Step 4: Drop firms with IPO after 2023-01-01
    # EQY_FUND_DT is returned as a string from Bloomberg; parse carefully
    def parse_date(s):
        if not s or s == "None":
            return None
        try:
            return datetime.strptime(str(s).strip(), "%Y-%m-%d")
        except ValueError:
            try:
                return datetime.strptime(str(s).strip(), "%m/%d/%Y")
            except ValueError:
                return None

    df["ipo_dt_parsed"] = df["ipo_date"].apply(parse_date)
    pre_filter = len(df)
    # Drop firms where ipo_dt_parsed is known and after cutoff
    df = df[df["ipo_dt_parsed"].isna() | (df["ipo_dt_parsed"] <= IPO_CUTOFF)]
    log.info(
        "Step 4: Dropped %d firms with IPO after %s. Remaining: %d.",
        pre_filter - len(df), IPO_CUTOFF.strftime("%Y-%m-%d"), len(df),
    )

    # Select final output columns
    output_cols = ["ticker", "name", "sector", "industry", "country", "ipo_date"]
    # Use bloomberg_ticker as fallback if ticker column is empty
    df["ticker"] = df["ticker"].fillna(df["bloomberg_ticker"])
    df = df[output_cols].copy()

    # Save
    os.makedirs(os.path.dirname(OUTPUT_PATH), exist_ok=True)
    df.to_csv(OUTPUT_PATH, index=False)
    log.info(
        "Saved %d firms to %s. Expected ~600-700 rows for KOSPI non-financials.",
        len(df), OUTPUT_PATH,
    )
    if len(df) < 300:
        log.warning(
            "Only %d firms in universe — significantly below expected ~600-700. "
            "Check BDS output and filter logic.",
            len(df),
        )


if __name__ == "__main__":
    main()
```
